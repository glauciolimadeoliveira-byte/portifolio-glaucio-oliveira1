// Aguarda o documento HTML carregar completamente antes de rodar os scripts
document.addEventListener('DOMContentLoaded', () => {

    /* ==========================================
       1. FUNCIONALIDADE: ALTERNADOR DE TEMA (CLARO/ESCURO)
       ========================================== */
    const themeToggleBtn = document.getElementById('theme-toggle');
    const bodyElement = document.body;

    themeToggleBtn.addEventListener('click', () => {
        // Altera a classe entre light-theme e dark-theme
        if (bodyElement.classList.contains('dark-theme')) {
            bodyElement.classList.remove('dark-theme');
            bodyElement.classList.add('light-theme');
            themeToggleBtn.textContent = '🌙 Modo Escuro';
        } else {
            bodyElement.classList.remove('light-theme');
            bodyElement.classList.add('dark-theme');
            themeToggleBtn.textContent = '☀️ Modo Claro';
        }
    });

    /* ==========================================
       2. FUNCIONALIDADE: VALIDAÇÃO DO FORMULÁRIO DE CONTATO
       ========================================== */
    const contactForm = document.getElementById('contact-form');

    contactForm.addEventListener('submit', (event) => {
        // Previne o envio padrão do formulário que recarregaria a página
        event.preventDefault();

        // Captura dos elementos e valores digitados pelo usuário
        const nameField = document.getElementById('name');
        const emailField = document.getElementById('email');
        const messageField = document.getElementById('message');

        const nameValue = nameField.value.trim();
        const emailValue = emailField.value.trim();
        const messageValue = messageField.value.trim();

        // Expressão Regular (Regex) para verificar a validação padrão de e-mail
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

        // Validação estruturada campo a campo
        if (nameValue === "") {
            alert("Por favor, preencha o campo Nome Completo.");
            nameField.focus();
            return;
        }

        if (emailValue === "") {
            alert("Por favor, preencha o campo E-mail.");
            emailField.focus();
            return;
        }

        if (!emailRegex.test(emailValue)) {
            alert("Por favor, informe um formato de e-mail válido (exemplo: usuario@dominio.com).");
            emailField.focus();
            return;
        }

        if (messageValue === "") {
            alert("Por favor, preencha o campo Mensagem.");
            messageField.focus();
            return;
        }

        // Se passar por todas as validações, simula o envio com sucesso
        alert("Mensagem enviada com sucesso!");

        // Reseta todos os campos do formulário para o estado original em branco
        contactForm.reset();
    });
});
